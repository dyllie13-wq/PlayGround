// SQFLexer.cs — tokenizes params, execVM, select, forEach...
public enum Tok { Ident, Number, String, LBrace, RBrace,
  LBracket, RBracket, LParen, RParen, Semi, Comma, Op, Eof }
public sealed class Token { public Tok Kind; public string Text; }
public static class SQFLexer {
  public static List<Token> Tokenize(string src) {
    var out_ = new List<Token>(); int i=0;
    while(i<src.Length){
      char c=src[i];
      if(char.IsWhiteSpace(c)){i++;continue;}
      if(c=='/'&&i+1<src.Length&&src[i+1]=='/'){
        while(i<src.Length&&src[i]!='\n')i++;continue;}
      if(c=='"'){int j=i+1;
        while(j<src.Length&&src[j]!='"')j++;
        out_.Add(new Token{Kind=Tok.String,
          Text=src.Substring(i+1,j-i-1)});i=j+1;continue;}
      if(char.IsDigit(c)||(c=='.'&&i+1<src.Length
        &&char.IsDigit(src[i+1]))){int j=i;
        while(j<src.Length&&"0123456789.".Contains(src[j]))j++;
        out_.Add(new Token{Kind=Tok.Number,
          Text=src.Substring(i,j-i)});i=j;continue;}
      if(char.IsLetter(c)||c=='_'){int j=i;
        while(j<src.Length&&(char.IsLetterOrDigit(src[j])
          ||src[j]=='_'))j++;
        out_.Add(new Token{Kind=Tok.Ident,
          Text=src.Substring(i,j-i)});i=j;continue;}
      // single-char tokens + ==, !=, &&, ||
      var two = i+1<src.Length?src.Substring(i,2):"";
      if(two=="=="||two=="!="||two==">="||two=="<="){
        out_.Add(new Token{Kind=Tok.Op,Text=two});i+=2;continue;}
      Tok k = c switch{'{'=>Tok.LBrace,'}'=>Tok.RBrace,
        '['=>Tok.LBracket,']'=>Tok.RBracket,
        '('=>Tok.LParen,')'=>Tok.RParen,
        ';'=>Tok.Semi,','=>Tok.Comma,_=>Tok.Op};
      out_.Add(new Token{Kind=k,Text=c.ToString()});i++;
    }
    out_.Add(new Token{Kind=Tok.Eof,Text=""});return out_;
  }
}

// SQFThreadManager.cs — cooperative, frame-budgeted
public sealed class SQFScheduler {
  readonly Queue<IEnumerator<double>> ready = new();
  public void Spawn(Func<IEnumerator<double>> fn)
    => ready.Enqueue(fn());
  public void Tick(double budgetMs=2.0){ // called per frame
    var t0=DateTime.UtcNow;
    int n=ready.Count;
    for(int k=0;k<n;k++){
      var co=ready.Dequeue();
      if(co.MoveNext()) ready.Enqueue(co); // yields sleep secs
      if((DateTime.UtcNow-t0).TotalMilliseconds>budgetMs)
        break;
    }
  }
}

public enum EdMode { Flycam3D, Map2D, Live }
public sealed class PlacedEntity {
  public string Id, Prefab, Faction="BLUFOR";
  public Vector3 Pos; public float Yaw, Health=1;
  public string VarName="", InitScript="";
}
public sealed class MissionEditor {
  public EdMode Mode=EdMode.Map2D;
  public List<PlacedEntity> Entities=new();
  public PlacedEntity PlaceEntity(string prefab,
    Vector3 p, bool alignGround=true){
    if(alignGround) p.Y=Terrain.SampleHeight(p.X,p.Z);
    var e=new PlacedEntity{Id=Guid.NewGuid().ToString("N"),
      Prefab=prefab,Pos=p};
    Entities.Add(e); Undo.Push(()=>Entities.Remove(e));
    return e;
  }
  public void MoveGizmo(PlacedEntity e,Vector3 d){
    e.Pos+=d; e.Pos.Y=Terrain.SampleHeight(e.Pos.X,e.Pos.Z);}
  public string ExportJson() =>
    JsonSerializer.Serialize(Entities,
      new JsonSerializerOptions{WriteIndented=true});
  public string ExportSqf(){ // init.sqf + description.ext
    var sb=new StringBuilder("// generated init.sqf\n");
    foreach(var e in Entities)
      sb.AppendLine(
        $"{e.VarName==""?"_v":e.VarName} = " +
        $"\"{e.Prefab}\" createVehicle [{e.Pos.X:F1}, " +
        $"{e.Pos.Z:F1}, 0];");
    return sb.ToString();
  }
}

// Ballistics.cs — semi-implicit Euler, per-tick
// F_drag = -k*|v|*v ; a = g + F_drag/m + wind
public static Vector3 Step(Vector3 p, Vector3 v,
  float dt, float k, Vector3 wind){
  var rel=v-wind; var drag=-k*rel.Length()*rel;
  v+=(new Vector3(0,-9.81f,0)+drag)*dt; p+=v*dt;
  return p;
}
